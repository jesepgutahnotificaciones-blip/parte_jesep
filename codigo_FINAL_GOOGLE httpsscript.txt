// ============================================================
// doGet
// ============================================================
function doGet(e) {
  const action = e.parameter.action || '';
  const callback = e.parameter.callback || '';
  
  console.log('=== doGet === action=' + action);

  try {
    let result;
    switch (action) {
      case 'login':
        result = handleLogin(e.parameter.username, e.parameter.password);
        break;
      case 'verifyDepKey':
        result = handleVerifyDepKey(e.parameter.username, e.parameter.depKey, e.parameter.dependencia);
        break;
      case 'getData':
        result = handleGetData(e.parameter.sheet || e.parameter.dependency || '');
        break;
      case 'saveData':
        result = handleSaveData(e.parameter.dependency || '', e.parameter);
        break;
      case 'getSummary':
        result = handleGetSummary(e.parameter.dependency || '', e.parameter.turno);
        break;
      case 'getTotalFuncionarios':
        result = handleGetTotalFuncionarios(e.parameter.dependency || '');
        break;
      case 'getCells':
        result = handleGetCells(e.parameter.sheet, e.parameter.cells);
        break;
      case 'getAuditoria':
        result = handleGetAuditoria(e.parameter.dependency || '');
        break;
      case 'getUsuarios':
        result = handleGetUsuarios();
        break;
      case 'getAllNovedades':
        result = handleGetAllNovedadesFast();
        break;
      default:
        result = { success: false, error: 'Accion no valida: ' + action };
    }

    if (callback && callback.trim() !== '') {
      return createJSONPResponse(result, callback);
    } else {
      return createJSONResponse(result);
    }

  } catch (error) {
    console.error('ERROR doGet:', error);
    const err = { success: false, error: error.toString() };
    if (callback && callback.trim() !== '') {
      return createJSONPResponse(err, callback);
    } else {
      return createJSONResponse(err);
    }
  }
}

// ============================================================
// doPost - REDIRIGE A doGet
// ============================================================
function doPost(e) {
  return doGet({ parameter: e.parameter, postData: e.postData });
}

// ============================================================
// HELPERS
// ============================================================
function createJSONResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function createJSONPResponse(data, callback) {
  const cb = callback.replace(/[^a-zA-Z0-9_]/g, '');
  return ContentService.createTextOutput(cb + '(' + JSON.stringify(data) + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
}

function getSpreadsheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

// ============================================================
// LOGIN
// ============================================================
function handleLogin(username, password) {
  if (!username || !password) {
    return { success: false, error: 'Usuario y contraseña requeridos' };
  }
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName('USUARIOS');
    if (!sheet) return { success: false, error: 'No existe la hoja USUARIOS' };
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const rowUsername = (row[0] || '').toString().trim();
      const rowPassword = (row[1] || '').toString().trim();
      if (rowUsername.toLowerCase() === username.toLowerCase().trim() && rowPassword === password.trim()) {
        return {
          success: true,
          usuario: {
            username: rowUsername,
            rol: (row[2] || '').toString().trim().toLowerCase(),
            dependencia: (row[3] || '').toString().trim(),
            token: (row[4] || '').toString().trim() || Utilities.getUuid()
          }
        };
      }
    }
    return { success: false, error: 'Usuario o contraseña incorrectos' };
  } catch (error) {
    return { success: false, error: 'Error del servidor: ' + error };
  }
}

// ============================================================
// VERIFICAR CLAVE DE DEPENDENCIA
// ============================================================
function handleVerifyDepKey(username, depKey, dependencia) {
  if (!username || !depKey || !dependencia) {
    return { success: false, error: 'Datos incompletos' };
  }
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName('USUARIOS');
    if (!sheet) return { success: false, error: 'No existe la hoja USUARIOS' };
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const rowUsername = (row[0] || '').toString().trim();
      if (rowUsername.toLowerCase() === username.toLowerCase().trim()) {
        const depName = dependencia.toString().trim().toUpperCase();
        const rowDependencia = (row[3] || '').toString().trim().toUpperCase();
        if (rowDependencia !== depName) {
          return { success: false, error: 'La dependencia del usuario no coincide' };
        }
        const rowDepKey = (row[4] || '').toString().trim();
        if (!rowDepKey) {
          return { success: false, error: 'El usuario no tiene clave de dependencia configurada' };
        }
        if (rowDepKey === depKey.toString().trim()) {
          return { success: true, message: 'Clave verificada' };
        } else {
          return { success: false, error: 'Clave de dependencia incorrecta' };
        }
      }
    }
    return { success: false, error: 'Usuario no encontrado' };
  } catch (error) {
    return { success: false, error: 'Error del servidor: ' + error.toString() };
  }
}

// ============================================================
// GET DATA
// ============================================================
function handleGetData(sheetName) {
  if (!sheetName) {
    return { success: false, error: 'Nombre de hoja no especificado' };
  }
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      return { success: false, error: 'Hoja no encontrada: ' + sheetName };
    }
    const lastRow = sheet.getLastRow();
    const rowsToRead = Math.min(1000, Math.max(0, lastRow - 1));
    if (rowsToRead === 0) {
      return { success: true, data: [], count: 0 };
    }
    const range = sheet.getRange(2, CONFIG.COL_GRADO, rowsToRead, 5);
    const values = range.getValues();
    const result = [];
    values.forEach(function (row, i) {
      const grado = row[0];
      const nombres = row[1];
      if ((grado && grado.toString().trim()) || (nombres && nombres.toString().trim())) {
        result.push({
          row: i + 2,
          grado: grado ? grado.toString().trim() : '',
          nombres: nombres ? nombres.toString().trim() : '',
          turno: row[2] ? row[2].toString().trim().toUpperCase() : '',
          novedad: row[3] ? row[3].toString().trim() : '',
          observacion: row[4] ? row[4].toString().trim() : ''
        });
      }
    });
    return { success: true, data: result, count: result.length };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// ============================================================
// SAVE DATA + ACTUALIZAR Registro_Cambios + AUDITORIA
// ============================================================
function handleSaveData(dependency, params) {
  const row = parseInt(params.row);
  const novedad = params.novedad || '';
  const observation = params.observation || '';
  const usuario = params.usuario || params.username || 'Sistema';
  
  console.log('=== handleSaveData V12 === dep=' + dependency + ' row=' + row + ' usuario=' + usuario);
  
  if (!row || isNaN(row) || row < 2) {
    return { success: false, error: 'Fila invalida: ' + params.row };
  }
  
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(dependency);
    if (!sheet) {
      return { success: false, error: 'Dependencia no encontrada: ' + dependency };
    }
    
    const lastRow = sheet.getLastRow();
    if (row > lastRow) {
      return { success: false, error: 'Fila ' + row + ' no existe. Ultima: ' + lastRow };
    }
    
    // Guardar en G y H
    sheet.getRange(row, CONFIG.COL_NOVEDAD).setValue(novedad);
    sheet.getRange(row, CONFIG.COL_OBS).setValue(observation);
    
    // V12: Actualizar Registro_Cambios
    try {
      actualizarRegistroCambios(ss, dependency, novedad);
    } catch (rcError) {
      console.log('⚠️ Error actualizando Registro_Cambios (no critico):', rcError);
    }
    
    // Registrar auditoria
    try {
      registrarAuditoria(usuario, 'GUARDAR_NOVEDAD', 
        'Fila ' + row + ' | Novedad: ' + novedad + ' | Dep: ' + dependency);
    } catch (auditError) {
      console.log('⚠️ Error auditoria (no critico):', auditError);
    }
    
    return { 
      success: true, 
      message: 'Datos guardados correctamente', 
      row: row, 
      novedad: novedad, 
      observation: observation 
    };
    
  } catch (error) {
    console.error('Error saveData:', error);
    return { success: false, error: 'Error al guardar: ' + error.toString() };
  }
}

// ============================================================
// V12: Actualizar contadores en Registro_Cambios
// ============================================================
function actualizarRegistroCambios(ss, dependency, novedad) {
  const rcSheet = ss.getSheetByName('Registro_Cambios');
  if (!rcSheet) {
    console.log('Registro_Cambios no existe, omitiendo actualizacion');
    return;
  }
  
  // Determinar turno desde el nombre de la hoja
  let turno = '';
  if (dependency.endsWith('_TURNO_A')) turno = 'A';
  else if (dependency.endsWith('_TURNO_B')) turno = 'B';
  else if (dependency.endsWith('_TURNO_C')) turno = 'C';
  
  if (!turno) {
    console.log('No se pudo determinar turno de:', dependency);
    return;
  }
  
  // Mapeo de novedades a filas en Registro_Cambios
  // Turno A: J8:J21, Turno B: Q8:Q21, Turno C: X8:X21
  const novedadesMap = {
    'OFICINA': 0, 'VACACIONES': 1, 'SERVICIO': 2, 'ESTADO DE GRAVIDEZ': 3,
    'EXCUSA TOTAL': 4, 'PERMISO': 5, 'CITA MEDICA': 6, 'LICENCIA DE MATERNIDAD': 7,
    'EXCUSADO DEL SERVICIO': 8, 'OTRA NOVEDAD': 9, 'RETARDAD@': 10, 'EXCUSA PARCIAL': 11,
    'CURSO DE ASCENSO': 12, 'COMISION': 13
  };
  
  const idx = novedadesMap[novedad];
  if (idx === undefined) {
    console.log('Novedad no mapeada:', novedad);
    return;
  }
  
  const fila = 8 + idx; // J8=0, J9=1, etc.
  let columna = '';
  if (turno === 'A') columna = 'J';
  else if (turno === 'B') columna = 'Q';
  else if (turno === 'C') columna = 'X';
  
  const celda = columna + fila;
  const valorActual = rcSheet.getRange(celda).getValue() || 0;
  rcSheet.getRange(celda).setValue(valorActual + 1);
  
  console.log('✅ Registro_Cambios actualizado:', celda, '=', valorActual + 1);
}

// V12: getAllNovedades OPTIMIZADO
// Lee solo columna G de cada hoja de turno, en lotes
// ============================================================
function handleGetAllNovedadesFast() {
  console.log('=== handleGetAllNovedadesFast V12 ===');
  
  const resultado = { A: {}, B: {}, C: {} };
  const novedadesList = [
    'S/N', 'OFICINA', 'VACACIONES', 'SERVICIO', 'ESTADO DE GRAVIDEZ',
    'EXCUSA TOTAL', 'PERMISO', 'CITA MEDICA', 'LICENCIA DE MATERNIDAD',
    'EXCUSADO DEL SERVICIO', 'OTRA NOVEDAD', 'RETARDAD@', 'EXCUSA PARCIAL',
    'CURSO DE ASCENSO', 'COMISION'
  ];
  novedadesList.forEach(n => { resultado.A[n] = 0; resultado.B[n] = 0; resultado.C[n] = 0; });
  
  try {
    const ss = getSpreadsheet();
    const sheets = ss.getSheets();
    
    // Filtrar hojas de turno
    const hojasTurno = [];
    sheets.forEach(function(s) {
      const name = s.getName();
      if (name.endsWith('_TURNO_A')) hojasTurno.push({ sheet: s, turno: 'A', name: name });
      else if (name.endsWith('_TURNO_B')) hojasTurno.push({ sheet: s, turno: 'B', name: name });
      else if (name.endsWith('_TURNO_C')) hojasTurno.push({ sheet: s, turno: 'C', name: name });
    });
    
    console.log('Hojas de turno encontradas:', hojasTurno.length);
    
    // Procesar en lotes de 10 para evitar timeout
    const BATCH_SIZE = 10;
    for (let i = 0; i < hojasTurno.length; i += BATCH_SIZE) {
      const batch = hojasTurno.slice(i, i + BATCH_SIZE);
      batch.forEach(function(item) {
        try {
          const lastRow = item.sheet.getLastRow();
          if (lastRow <= 1) return;
          
          // Leer TODA la columna G de una vez (mucho mas rapido)
          const colG = item.sheet.getRange(2, CONFIG.COL_NOVEDAD, lastRow - 1, 1).getValues();
          
          colG.forEach(function(row) {
            const nov = (row[0] || '').toString().trim();
            if (nov && resultado[item.turno][nov] !== undefined) {
              resultado[item.turno][nov]++;
            }
          });
          
        } catch (e) {
          console.log('Error hoja', item.name, ':', e.message);
        }
      });
    }
    
    console.log('Novedades A:', JSON.stringify(resultado.A).substring(0, 100));
    return { success: true, novedades: resultado };
    
  } catch (error) {
    console.error('Error getAllNovedadesFast:', error);
    return { success: false, error: error.toString() };
  }
}

// ============================================================
// GET SUMMARY
// ============================================================
function handleGetSummary(dependency, turno) {
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(dependency);
    if (!sheet) return { success: false, error: 'Dependencia no encontrada' };
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return { success: true, summary: [] };
    const values = sheet.getRange(2, 1, lastRow - 1, 8).getValues();
    let filteredData = values;
    if (turno) {
      filteredData = values.filter(function (row) {
        return (row[CONFIG.COL_TURNO - 1] || '').toString().trim().toUpperCase() === turno.toUpperCase();
      });
    }
    const total = filteredData.length;
    const conNovedad = filteredData.filter(function (r) {
      return (r[CONFIG.COL_NOVEDAD - 1] || '').toString().trim() !== '';
    }).length;
    const sinNovedad = total - conNovedad;
    const novedadesCount = {};
    filteredData.forEach(function (row) {
      const nov = (row[CONFIG.COL_NOVEDAD - 1] || '').toString().trim();
      if (nov) novedadesCount[nov] = (novedadesCount[nov] || 0) + 1;
    });
    const summary = [];
    summary.push({ section: 'header', label: 'RESUMEN ' + dependency + (turno ? ' - TURNO ' + turno : ''), value: '' });
    summary.push({ section: 'normal', label: 'Total Funcionarios', value: total });
    summary.push({ section: 'normal', label: 'Con Novedad', value: conNovedad });
    summary.push({ section: 'normal', label: 'Sin Novedad', value: sinNovedad });
    if (Object.keys(novedadesCount).length > 0) {
      summary.push({ section: 'novedad-header', label: 'DETALLE DE NOVEDADES', value: '' });
      for (const [novedad, count] of Object.entries(novedadesCount)) {
        summary.push({ section: 'novedad-item', label: novedad, value: count });
      }
    }
    return { success: true, summary: summary };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// GET CELLS
// ============================================================
function handleGetCells(sheetName, cellsParam) {
  if (!sheetName || !cellsParam) {
    return { success: false, error: 'Parametros sheet y cells requeridos' };
  }
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) return { success: false, error: 'Hoja no encontrada: ' + sheetName };
    const celdas = cellsParam.split(',').map(function(c) { return c.trim(); });
    const valores = {};
    celdas.forEach(function(celda) {
      if (!celda) return;
      try { valores[celda] = sheet.getRange(celda).getValue(); } catch(e) { valores[celda] = ''; }
    });
    return { success: true, values: valores };
  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

      });
    });
    return { success: true, data: result };

    return { success: false, error: error.toString() };
  }
}

function crearHojaDependencia(nombreDependencia, claveDependencia) {
  const ss = getSpreadsheet();
  if (ss.getSheetByName(nombreDependencia)) {
    return 'La hoja ' + nombreDependencia + ' ya existe';
  }
  const sheet = ss.insertSheet(nombreDependencia);
  const headers = ['ID', '', '', 'GRADO', 'APELLIDOS Y NOMBRES', 'TURNO', 'NOVEDAD', 'OBSERVACION'];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers])
    .setFontWeight('bold').setBackground('#01592F').setFontColor('#FFFFFF');
  sheet.setColumnWidth(1, 50); 
  sheet.setColumnWidth(4, 80);
  sheet.setColumnWidth(5, 250);
  sheet.setColumnWidth(6, 60);
  sheet.setColumnWidth(7, 150);
  sheet.setColumnWidth(8, 200);
  const novedades = [
    'S/N','OFICINA','VACACIONES','SERVICIO','ESTADO DE GRAVIDEZ',
    'EXCUSA TOTAL','PERMISO','CITA MEDICA','LICENCIA DE MATERNIDAD',
    'EXCUSADO DEL SERVICIO','OTRA NOVEDAD','RETARDAD@','EXCUSA PARCIAL',
    'CURSO DE ASCENSO','COMISION'
  ];
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(novedades, true)
    .setAllowInvalid(false)
    .setHelpText('Seleccione una novedad valida')
    .build();
  sheet.getRange('G2:G1000').setDataValidation(rule);
  sheet.setFrozenRows(1);
  return 'Hoja ' + nombreDependencia + ' creada exitosamente';
}

function mostrarUsuariosYClaves() {
  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName('USUARIOS');
    if (!sheet) { SpreadsheetApp.getUi().alert('No existe hoja USUARIOS'); return; }
    const data = sheet.getDataRange().getValues();
    let msg = 'USUARIOS Y CLAVES DE DEPENDENCIA (Col E):\\n\\n';
    for (let i = 1; i < data.length; i++) {
      const user = (data[i][0] || '').toString().trim();
      const dep = (data[i][3] || '').toString().trim();
      const clave = (data[i][4] || '').toString().trim();
      if (user) {
        msg += user + ' | Dep: ' + dep + ' | Clave: ' + (clave || '⚠️ SIN CLAVE') + '\\n';
      }
    }
    SpreadsheetApp.getUi().alert(msg);
  } catch(e) {
    SpreadsheetApp.getUi().alert('Error: ' + e.toString());
  }
}

function listarDependencias() {
  const ss = getSpreadsheet();
  const sheets = ss.getSheets();
  const nombres = sheets.map(function (s) { return s.getName(); })
                        .filter(function (n) { return n !== CONFIG.USERS_SHEET; });
  SpreadsheetApp.getUi().alert('Dependencias: ' + nombres.join(', '));
}
